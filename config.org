** Clasp Integration
Wrapper functions for the Google clasp CLI. Run from any file inside the project — they use projectile to find the project root.

#+begin_src emacs-lisp
(defun rh/clasp-in-project (cmd &optional buffer-name async)
  "Run clasp CMD from current projectile project root.
If ASYNC is non-nil, run as async-shell-command in BUFFER-NAME."
  (let ((default-directory (or (projectile-project-root)
                               default-directory)))
    (if async
        (async-shell-command cmd (or buffer-name "*clasp*"))
      (compile cmd))))

(defun rh/clasp-push ()
  "Push local Apps Script files to Google."
  (interactive)
  (rh/clasp-in-project "clasp push"))

(defun rh/clasp-push-watch ()
  "Auto-push to Google on every save (runs in background buffer)."
  (interactive)
  (rh/clasp-in-project "clasp push --watch" "*clasp-watch*" t))

(defun rh/clasp-pull ()
  "Pull Apps Script files from Google."
  (interactive)
  (rh/clasp-in-project "clasp pull"))

(defun rh/clasp-open ()
  "Open the Apps Script project in the browser."
  (interactive)
  (rh/clasp-in-project "clasp open" nil nil))

(defun rh/clasp-logs ()
  "Tail Apps Script execution logs."
  (interactive)
  (rh/clasp-in-project "clasp logs --watch" "*clasp-logs*" t))

(defun rh/clasp-deployments ()
  "List Apps Script deployments."
  (interactive)
  (rh/clasp-in-project "clasp deployments"))

(defun rh/clasp-status ()
  "Show clasp status (which files will be pushed/pulled)."
  (interactive)
  (rh/clasp-in-project "clasp status"))

* Claude Code
Automatically exports skill org files to SKILL.md with YAML frontmatter.
Only triggers for files with names ending in -skill.org.
Reads frontmatter values from #+SKILL_NAME and #+SKILL_DESC keywords.
#+begin_src emacs-lisp
(require 'ox-md)

(defun my/export-claude-skill ()
  "Export a -skill.org file to SKILL.md with YAML frontmatter.
 Only runs when the saved file ends with -skill.org."
  (when (and (eq major-mode 'org-mode)
             (buffer-file-name)
             (string-suffix-p "-skill.org"
                              (file-name-nondirectory (buffer-file-name))))
    (let* ((org-file (buffer-file-name))
           (dir (file-name-directory org-file))
           (tmp-file (expand-file-name "SKILL.md.tmp" dir))
           (output-file (expand-file-name "SKILL.md" dir))
           (skill-name (or (save-excursion
                             (goto-char (point-min))
                             (when (re-search-forward
                                    "^#\\+SKILL_NAME:\\s-*\\(.+\\)" nil t)
                               (match-string 1)))
                           "unnamed-skill"))
           (skill-desc (or (save-excursion
                             (goto-char (point-min))
                             (when (re-search-forward
                                    "^#\\+SKILL_DESC:\\s-*\\(.+\\)" nil t)
                               (match-string 1)))
                           "No description")))
      (org-export-to-file 'md tmp-file nil nil nil nil nil)
      (let ((md-content (with-temp-buffer
                          (insert-file-contents tmp-file)
                          (buffer-string)))
            (frontmatter (format "---\nname: %s\ndescription: %s\n---\n\n"
                                 skill-name skill-desc)))
        (with-temp-file output-file
          (insert frontmatter)
          (insert md-content))
        (delete-file tmp-file)
        (message "Claude skill exported to %s" output-file)))))

(add-hook 'after-save-hook #'my/export-claude-skill)
#+end_src

* Gptel 
gptel~ is a simple, no-frills Large Language Model (LLM) client for Emacs. It lets you chat with models like ChatGPT, Claude, Gemini, and
local models (Ollama, llama.cpp) directly from any Emacs buffer.

Unlike heavier "AI assistant" packages, gptel treats LLM interaction as plain text in ordinary buffers. You can send a region, the buffer
above point, or a dedicated chat buffer — and the response is just inserted as text you can edit, save, and version-control.

CBorg is reached at =https://api.cborg.lbl.gov=, which is fronted by Cloudflare and can block clients on high request rate — exactly what
tool-call loops produce, since every round-trip is a fresh request. On the LBL network (VPN, employee wifi, ethernet) the =CBorg-Local=
backend routes directly to =https://api-local.cborg.lbl.gov= and skips Cloudflare entirely.

Model IDs verified against =/v1/models= on 2026-08-23 via =M-x my/cborg-list-models=.

- *Prefix is the hosting cloud, not the vendor.* Hence both =amazon/claude-opus-4-8= and =google/claude-opus-4-8=. Unprefixed IDs
  (=claude-fable-5=, =cborg-coder=) are the gateway's own aliases and are what this config uses — no prefix means "let CBorg route it".
- *Reasoning effort is part of the model ID*, not a request parameter: =-low=, =-medium=, =-high=, =-xhigh=, =-max=, =-ultra=. So
  =claude-fable-5-high= is a genuinely different model string from =claude-fable-5=, and no =:request-params= are needed to select effort.
  The ladder is not uniform: most models top out at =-max= or =-xhigh=, and =-ultra= currently exists only on fable-5.
- *Rolling vs pinned.* =claude-sonnet= / =gemini-pro= / =gpt= track the current best in their class and will silently upgrade under you.
  =claude-fable-5= / =gemini-3.1-pro= are pinned. Both kinds are in the list below; prefer rolling for daily use, pinned when you need
  reproducibility.
- *=-compact= / =-compressed= suffixes* exist on nearly every model. Purpose unverified — presumably server-side context compaction.
  Check the CBorg docs before using; not wired up here.
- LBL-hosted models (=cborg-*=, =gemma-4-*=, =gpt-oss-*=, =nomic-embed-*=) are the free/local tier. Everything else bills against a
  commercial provider.

  "Left-aligned gptel header line with clickable elements."
  (condition-case err

       (propertize
        (buttonize (concat "[" (gptel--model-name gptel-model) "]")

                           (or (car-safe (rassoc gptel--system-message gptel-directives))
                               (gptel--describe-directive gptel--system-message 15)))
                   (lambda (&rest _) (gptel-system-prompt)))
        'mouse-face 'highlight 'help-echo "System message for session")
       (when (and gptel-use-tools gptel-tools)
         (concat " "
                 (propertize
                  (buttonize (format "[%dt]" (length gptel-tools))
                             (lambda (&rest _)
                               (run-at-time 0 nil
                                            (lambda () (call-interactively #'gptel-tools)))))
                  'mouse-face 'highlight 'help-echo "Select tools")))
       (when gptel-context
         (concat " "
                 (propertize
                  (buttonize (format "[%d ctx]" (length gptel-context))
                             (lambda (&rest _)
                               (require 'gptel-context)
                               (gptel-context--buffer-setup)))
                  'mouse-face 'highlight 'help-echo "Active gptel context")))
       (and-let* ((idx (car-safe gptel--token-usage-strings))
                  (entry (or (nth (1+ idx) gptel--token-usage-strings) "[usage...]"))
                  (noinfo "[No info]"))
         (concat " "
                 (buttonize entry
                            (lambda (strings)
                              (and (car-safe strings)
                                   (cl-callf (lambda (pos) (% (1+ pos) 2)) (car strings))))
                            gptel--token-usage-strings
                            (concat "Token usage (C = cached tokens)\nLast request: "
                                    (or (cadr gptel--token-usage-strings) noinfo)
                                    "\nThis buffer:  "
                                    (or (caddr gptel--token-usage-strings) noinfo))))))
    (error (propertize (format " [hdr: %s]" (error-message-string err))
                       'face 'error))))

            (error (message "Failed to register MCP tool %s: %s"
                            (plist-get tool :name) err))))
        (message "Registered %d MCP tools with gptel" count)
        count)))

  ;; Start every server in `mcp-hub-servers' and register their tools
  ;; with gptel once they're up. Then — now that the tools exist —
  ;; load and apply the gptel presets that reference them.
  (let ((starter
         (lambda ()
           (mcp-hub-start-all-server
            (lambda ()
              (run-with-timer
               2 nil
               (lambda ()
                 (when (fboundp 'mcp-hub-register-all-tool-to-gptel)
                   (mcp-hub-register-all-tool-to-gptel)
                   (with-eval-after-load 'gptel
                     (setq gptel-tools
                           (cl-remove-duplicates
                            (append gptel-tools
                                    (gptel-get-tool '("mcp-global-memory"))) ;; auto-enable global memory tools as defaults
                            :test #'equal))
                     ;; ---- Now MCP tools exist: load + apply gptel presets ----
                     ;; `my/gptel-load-and-apply-presets' sets
                     ;; `my/gptel-presets-applied' itself, which tells the
                     ;; gptel fallback timer to stand down.
                     (when (fboundp 'my/gptel-load-and-apply-presets)
                       (my/gptel-load-and-apply-presets))))))))))) 
    (if after-init-time
        (funcall starter)
      (add-hook 'after-init-hook starter))))

;;;----------------------------------------------------------------------
;;; Project-local memory: lazy startup helpers
;;;----------------------------------------------------------------------

(defvar my/project-mcp-started nil
  "List of project roots whose memory MCP server has already been started.
Used to avoid spawning duplicate processes for the same project.")

(defun my/project-mcp-server-name (proj-root)
  "Derive a stable, human-readable server name from PROJ-ROOT."
  (format "project-memory-%s"
          (file-name-nondirectory (directory-file-name proj-root))))

(defun my/ensure-project-mcp-memory ()
  "Start a project-local memory MCP server for the current project, if needed.

Looks up the current project via `project-current'. If found, and we
haven't already started a server for that root, registers a new entry
in `mcp-hub-servers', launches it, and registers its tools with gptel
once the connection is established.

Safe to call repeatedly: it's a no-op for projects that have already
been initialised, and a no-op when not inside a project."
  (interactive)
  (when-let* ((proj      (project-current nil))
              (proj-root (expand-file-name
                          (if (fboundp 'project-root)
                              (project-root proj)
                            (cdr proj)))))
    (unless (member proj-root my/project-mcp-started)
      (let* ((server-name   (my/project-mcp-server-name proj-root))
             (memory-file   (expand-file-name ".mcp-project-memory.json"
                                              proj-root))
             (server-config `(:command "npx"
                              :args    ("-y" "@modelcontextprotocol/server-memory")
                              :env     (:MEMORY_FILE_PATH ,memory-file
			      :npm_config_update_notifier "false"
                              :NODE_OPTIONS "--no-warnings"))))

        ;; Record this project *before* starting the server, so
        ;; concurrent calls don't race and double-start.
        (push proj-root my/project-mcp-started)

        ;; Register the per-project server with the hub.
        (add-to-list 'mcp-hub-servers (cons server-name server-config))

        (condition-case err
            (mcp-hub-start-all-server
             (lambda ()
               (run-with-timer
                1 nil
                (lambda ()
                  (when (fboundp 'mcp-hub-register-all-tool-to-gptel)
                    (mcp-hub-register-all-tool-to-gptel))
                  ;; Auto-enable THIS project's memory tools in gptel.
                  (with-eval-after-load 'gptel
                    (let ((cat (concat "mcp-"
                                       (my/project-mcp-server-name proj-root))))
                      (setq gptel-tools
                            (cl-remove-duplicates
                             (append gptel-tools
                                     (gptel-get-tool (list cat)))
                             :test #'equal))))
                  (message "MCP: project memory server ready for %s"
                           proj-root)))))
          (error
           ;; Roll back the started-list entry so a retry is possible.
           (setq my/project-mcp-started
                 (delete proj-root my/project-mcp-started))
           (signal (car err) (cdr err))))))))

(defun my/reset-project-mcp ()
  "Forget which projects have had their MCP memory server started.
Does not stop any running servers; use `mcp-hub' for that."
  (interactive)
  (setq my/project-mcp-started nil)
  (message "Project MCP started-list cleared."))

(defun neotree--sibling-window ()
  "Return the editing window adjacent to NeoTree.
Prefers the window on the side the tree opens toward; falls back to the
first ordinary window on the frame.  Returns nil if there is none."
  (let* ((neo-win (selected-window))
         (dir (if (eq neo-window-position 'right) 'left 'right))
         (adjacent (window-in-direction dir neo-win)))
    (or (and (window-live-p adjacent)
             (not (neotree--window-p adjacent))
             adjacent)
        (seq-find (lambda (w)
                    (and (not (eq w neo-win))
                         (not (neotree--window-p w))
                         (not (window-parameter w 'window-side))))
                  (window-list nil nil (frame-first-window))))))

(defun neotree--make-target-window ()
  "Create and return a window to the right of the frame's main area."
  (let ((main (if (fboundp 'window-main-window)
                  (window-main-window)
                (frame-root-window))))
    (split-window main nil 'right)))

(defun neotree-enter-or-click-in-winum (&optional arg event)
  "Open the NeoTree file at point in another window.
With no prefix ARG, use the window adjacent to NeoTree, creating one if
none exists.  With a numeric prefix, use that winum window.  With
`C-u', prompt for a window number.  EVENT is the mouse event when
invoked by clicking.  Directories fall through to `neotree-enter'."
  (interactive (list current-prefix-arg last-nonmenu-event))
  (when (mouse-event-p event)
    (mouse-set-point event))
  (let ((file (neo-buffer--get-filename-current-line)))
    (if (not (and file (not (file-directory-p file))))
        (neotree-enter)
      (let* ((sibling (neotree--sibling-window))
             (default (and sibling (winum-get-number sibling)))
             (num (cond ((integerp arg) arg)
                        ((consp arg) (read-number "Window number: " default))
                        (t default)))
             (win (cond ((and (integerp num) (> num 0))
                         (winum-get-window-by-number num))
                        (t sibling))))
        (cond
         ((and (integerp num) (null win))
          (message "Window %d does not exist!" num))
         ((eq win (selected-window))
          (message "That's the NeoTree window itself!"))
         (t
          (let ((created (null win)))
            (select-window (or win (neotree--make-target-window)))
            (find-file file)
            (when created
              (balance-windows)))))))))

(defun my/gptel-magit-commit-message ()
  "Generate a git commit message based on the staged diff using gptel."
  (interactive)
  (let ((diff (with-temp-buffer
                (magit-git-insert "diff" "--cached")
                (buffer-string)))
        (buf (current-buffer)))
    (when (or (null diff)
              (string-empty-p (string-trim diff))
              (not (string-match-p "^[+-]" diff)))
      (user-error "No staged changes to commit (did you `git add`?)"))
    ;; Check for existing user-written content (ignoring comment lines and blanks)
    (when (save-excursion
            (goto-char (point-min))
            (let ((has-content nil))
              (while (and (not has-content) (not (eobp)))
                (let ((line (buffer-substring-no-properties
                             (line-beginning-position) (line-end-position))))
                  (unless (or (string-empty-p (string-trim line))
                              (string-prefix-p "#" (string-trim line)))
                    (setq has-content t)))
                (forward-line 1))
              has-content))
      (unless (yes-or-no-p "Commit buffer already has a message. Overwrite? ")
        (user-error "Aborted; existing message kept")))
    (message "Generating commit message...")
    (gptel-request
     (concat "Write a clear, concise git commit message for the following diff. "
             "Use the imperative mood for the subject line. "
             "Output ONLY the commit message (a short subject line, a blank line, and a brief body if needed), "
             "with no extra markdown formatting, quotes, or explanations.\n\n"
             diff)
     :callback
     (lambda (response info)
       (cond
        ;; Ignore reasoning chunks and other non-string payloads
        ((and (consp response) (eq (car response) 'reasoning))
         nil)
        ((not (stringp response))
         (message "gptel returned non-text response: %S" response))
        ((string-empty-p (string-trim response))
         (message "gptel returned empty response. Info: %S" info))
        (t
         (with-current-buffer buf
           ;; Clear any existing non-comment content before inserting
           (save-excursion
             (goto-char (point-min))
             (let ((kill-end
                    (save-excursion
                      (goto-char (point-min))
                      (if (re-search-forward "^#" nil t)
                          (line-beginning-position)
                        (point-max)))))
               (delete-region (point-min) kill-end)))
           (goto-char (point-min))
           (insert (string-trim response))
           (insert "\n")
           (message "Commit message generated!"))))))))

(with-eval-after-load 'git-commit
  (define-key git-commit-mode-map (kbd "C-c C-m") #'my/gptel-magit-commit-message))

;;;; Monitor discovery

(defun my/layout--area (monitor)
  "Usable pixel area (X Y W H) of MONITOR."
  (or (alist-get 'workarea monitor)
      (alist-get 'geometry monitor)))

(defun my/layout--portrait-monitor ()
  "First monitor taller than it is wide, or nil."
  (seq-find (lambda (m)
              (pcase-let ((`(,_ ,_ ,w ,h) (my/layout--area m)))
                (> h w)))
            (display-monitor-attributes-list)))

(defun my/layout--landscape-monitor ()
  "Widest monitor that is wider than it is tall, or nil."
  (car (sort (seq-filter (lambda (m)
                           (pcase-let ((`(,_ ,_ ,w ,h) (my/layout--area m)))
                             (>= w h)))
                         (display-monitor-attributes-list))
             (lambda (a b) (> (nth 2 (my/layout--area a))
                              (nth 2 (my/layout--area b)))))))

(defun my/layout--coord (n)
  "Frame-parameter position spec for absolute pixel coordinate N.
Bare negative integers mean \"offset from the far edge\", so a genuinely
negative absolute coordinate must use the (+ N) list form."
  (if (< n 0) (list '+ n) n))

(defun my/layout--display-of (frame)
  "Return the display name FRAME lives on, or nil."
  (frame-parameter frame 'display))

(defun my/layout-window (buffer &optional frame)
  "Return a live window showing BUFFER, or nil.
BUFFER is a buffer or a buffer name.  FRAME limits the search to that
frame; otherwise all visible frames are searched."
  (when-let ((buf (get-buffer buffer)))
    (get-buffer-window buf (or frame 'visible))))

(defun my/layout-frame (role)
  "Return the live frame tagged with ROLE, or nil."
  (or (seq-find (lambda (f)
                  (and (frame-live-p f)
                       (eq (frame-parameter f 'my/layout-role) role)))
                (frame-list))
      (and (eq role 'portrait)
           (when-let* ((b (my/layout-gptel-buffer))
                       (w (my/layout-window b)))
             (window-frame w)))))

(defun my/layout--place (frame monitor)
  "Move FRAME onto MONITOR and maximize it there."
  (when (and (frame-live-p frame) monitor)
    (pcase-let ((`(,x ,y ,_ ,_) (my/layout--area monitor)))
      (set-frame-parameter frame 'fullscreen nil)
      (redisplay t)
      (modify-frame-parameters frame
                               `((left . ,(my/layout--coord x))
                                 (top  . ,(my/layout--coord y))))
      (redisplay t)
      (set-frame-parameter frame 'fullscreen 'maximized))))

(defun my/layout--build-portrait-windows (frame)
  "Split FRAME into a general-purpose window above, gptel below.
An existing non-agent buffer in the frame is kept in the top window."
  (with-selected-frame frame
    (let* ((keep (seq-find (lambda (b)
                             (not (eq b (my/layout-gptel-buffer))))
                           (mapcar #'window-buffer (window-list frame 'no-mini))))
           (top-buffer (or keep
                           (get-buffer-create my/layout-portrait-top-buffer))))
      (delete-other-windows)
      (let ((bottom (split-window-below)))
        (switch-to-buffer top-buffer)
        (with-selected-window bottom
          (my/layout--show-gptel))))))

(defun my/layout--windows-ok-p (frame)
  "Non-nil if FRAME is split in two with the agent in the lower window."
  (let ((windows (window-list frame 'no-mini))
        (agent (my/layout-gptel-buffer)))
    (and (= (length windows) 2)
         (buffer-live-p agent)
         (when-let ((w (my/layout-window agent frame)))
           (> (window-top-line w) 0)))))

                (string-prefix-p "*gptel-agent:" (buffer-name b)))

  (if (buffer-live-p my/layout--gptel-buffer)
      my/layout--gptel-buffer

      (setq my/layout--gptel-buffer
            (when (= (length candidates) 1) (car candidates))))))

  (let ((buf (my/layout-gptel-buffer)))
    (if (buffer-live-p buf)
        (switch-to-buffer buf)
      (let* ((before (my/layout--agent-buffers))
             (display-buffer-overriding-action '((display-buffer-same-window))))

                (switch-to-buffer new))

           ((my/layout--agent-buffers)
            (setq my/layout--gptel-buffer (car (my/layout--agent-buffers)))

(defun my/layout-apply (&optional rebuild)
  "Place the layout frames on their monitors.

With prefix argument REBUILD, also reset the window splits and buffers.
Without it, only frame geometry is corrected, and windows that are
already arranged correctly are left alone."
  (interactive "P")
  (unless (display-graphic-p)
    (user-error "No graphical display"))
  (let ((my/layout--busy t)
        (portrait-mon (my/layout--portrait-monitor))
        (landscape-mon (my/layout--landscape-monitor)))

    ;; Main frame.
    (let ((main (or (my/layout-frame 'main) (selected-frame))))
      (set-frame-parameter main 'my/layout-role 'main)
      (if landscape-mon
          (my/layout--place main landscape-mon)
        (set-frame-parameter main 'fullscreen 'maximized))
      (when rebuild
        (with-selected-frame main
          (delete-other-windows)
          (split-window-right))))

    ;; Portrait frame.
    (if (not portrait-mon)
        (message "my/layout: no portrait monitor detected")
      (let ((frame (my/layout-frame 'portrait))
            (fresh nil))
        (unless frame

        (when (or fresh rebuild (not (my/layout--windows-ok-p frame)))
          (my/layout--build-portrait-windows frame))))
    (message "my/layout: %s" (if rebuild "rebuilt" "repositioned"))))

(defun my/layout-startup ()
  "Build the layout once, as soon as a graphical frame exists.
Deferred, because in daemon mode this runs before the client frame is
fully realised."
  (when (and (not my/layout--started)
             (display-graphic-p))
    (setq my/layout--started t)
    (let ((frame (selected-frame)))
      (run-at-time
       0.5 nil
       (lambda ()
         (if (not (frame-live-p frame))
             ;; Client vanished before we ran; allow a later attempt.
             (setq my/layout--started nil)
           (with-selected-frame frame
             (condition-case err
                 (my/layout-apply t)
               (error
                (setq my/layout--started nil)
                (message "my/layout: %S" err))))))))))
